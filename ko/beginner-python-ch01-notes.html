1. / 와 //

8 / 2   → 4.0  # 8 을 2 로 나눈 값은 얼마예요?
8 // 2  → 4    # 8 을 2 로 나눈 몫은 얼마예요?
7 / 2   → 3.5  # 7 을 2 로 나눈 값은 얼마예요?
7 // 2  → 3    # 7 을 2 로 나눈 몫은 얼마예요?

핵심 한 줄: /는 소수점 유지, //는 버림.

2. import와 math

math.sqrt(2)  → NameError: name 'math' is not defined. Did you forget to import 'math'?  # math 의 sqrt 로 2 의 제곱근은 얼마예요?
from math import sqrt                                                                    # math 에서 sqrt 도구만 가져와 주세요
sqrt(2)       → 1.4142135623730951                                                       # 2 의 제곱근은 얼마예요?
import math as m                                                                         # math 도구 상자를 m 이라는 이름으로 가져와 주세요
m.sqrt(2)     → 1.4142135623730951                                                       # m 의 sqrt 로 2 의 제곱근은 얼마예요?

3. ** 거듭제곱

3 ** 2   → 9             # 3 의 2제곱은 얼마예요?
3 ** 3   → 27            # 3 의 3제곱은 얼마예요?
2 ** 10  → 1024          # 2 의 10제곱은 얼마예요?
550 ^ 2  → 548           # 550 과 2 를 ^ 로 연산하면 얼마예요?
math.hypot(3, 4)  → 5.0  # 가로 3 세로 4 인 직각삼각형의 빗변은 얼마예요?

4. 리스트 인덱싱·슬라이싱

s = ['시청', '을지로입구', '을지로3가', '을지로4가', '동대문역사문화공원', '신당', '상왕십리', '왕십리']         # s 에 역 이름 8개의 목록을 담아 주세요
s[0]     → '시청'                                                                 # s 의 0번 원소는 무엇인가요?
s[-1]    → '왕십리'                                                                # s 의 맨 뒤 원소는 무엇인가요?
s[1:4]   → ['을지로입구', '을지로3가', '을지로4가']                                          # s 의 1번부터 4번 앞까지 잘라내면 어떻게 되나요?
s[::2]   → ['시청', '을지로3가', '동대문역사문화공원', '상왕십리']                                 # s 를 두 칸씩 건너뛰며 고르면 어떻게 되나요?
s[::-1]  → ['왕십리', '상왕십리', '신당', '동대문역사문화공원', '을지로4가', '을지로3가', '을지로입구', '시청']  # s 를 거꾸로 읽으면 어떻게 되나요?

5. 추가·삭제

f = ['mother', 'father']                                                              # f 에 'mother', 'father' 목록을 담아 주세요
f.append('me')              → ['mother', 'father', 'me']                              # f 의 맨 뒤에 'me' 를 붙여 주세요
f.extend(['sister', 'dog']) → ['mother', 'father', 'me', 'sister', 'dog']             # f 의 맨 뒤에 ['sister', 'dog'] 를 풀어서 붙여 주세요
f.insert(1, 'grandma')      → ['mother', 'grandma', 'father', 'me', 'sister', 'dog']  # f 의 1번 자리에 'grandma' 를 끼워 넣어 주세요
f.remove('dog')             → ['mother', 'grandma', 'father', 'me', 'sister']         # f 에서 'dog' 를 찾아 지워 주세요
f.pop()                     → 'sister'                                                # f 의 맨 뒤 원소를 꺼내면 무엇이 나오나요?
f.pop(1)                    → 'grandma'                                               # f 의 1번 원소를 꺼내면 무엇이 나오나요?
del f[0]                    → ['father', 'me']                                        # f 의 0번 원소를 지워 주세요

6. 메서드와 키워드

f.pop(0)                                                                       # f 의 0번 원소를 꺼내면 무엇이 나오나요?
del f[0]                                                                       # f 의 0번 원소를 지워 주세요
'hello'.upper()  → 'HELLO'                                                     # 'hello' 를 대문자로 바꾸면 어떻게 되나요?
'hello'.append('!')  → AttributeError: 'str' object has no attribute 'append'  # 'hello' 의 맨 뒤에 '!' 를 붙여 주세요

7. input()과 int()

num = int(input())                                                        # num 에 입력받은 글자를 정수로 바꿔 담아 주세요
print(num * num)                                                          # num 곱하기 num 을 화면에 보여 주세요
'3' * '3'  → TypeError: can't multiply sequence by non-int of type 'str'  # '3' 곱하기 '3' 은 얼마예요?
'3' * 3    → '333'                                                        # '3' 곱하기 3 은 얼마예요?
