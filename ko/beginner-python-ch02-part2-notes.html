1. and / or

# and: 둘 다 참이어야 참 / or: 하나라도 참이면 참
>>> s = 'banana'                      # 변수 s 에 문자열 'banana' 를 담아요
>>> if 'a' in s and 'b' in s:         # s 에 a 와 b 가 모두 있는지 따져요
...     print('a도 있고 b도 있어요')  # 둘 다 있으면 이 문장을 출력해요
...
a도 있고 b도 있어요
>>> a, b = 3, 0                       # 변수 a 에 3, b 에 0 을 한꺼번에 담아요
>>> (a * b) > 0 and (a / b) > 0       # 왼쪽이 False라서 오른쪽 나눗셈은 안 해요
False

2. for 문

# 원소를 앞에서부터 하나씩 x에 꺼낸다
>>> family = ['mother', 'father', 'sister']  # 변수 family 에 문자열 리스트를 담아요
>>> for x in family:                         # family 의 원소를 하나씩 꺼내 x 에 담아요
...     print(x, len(x))                     # x 와 x 의 글자 수를 함께 출력해요
...
mother 6
father 6
sister 6

3. range()

# range(시작, 끝): 끝은 빠진다
>>> list(range(2, 7))      # 2 부터 6 까지 숫자를 리스트로 만들어요
[2, 3, 4, 5, 6]
>>> for i in range(4, 8):  # 4 부터 7 까지 숫자를 하나씩 i 에 담아요
...     print(i)           # i 를 출력해요
...
4
5
6
7
