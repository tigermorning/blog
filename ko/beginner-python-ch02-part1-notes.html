무한 루프와 break

# filename: ten.py
max = 10                           # max 에 10 을 담아 주세요

while True:                        # True 인 동안 아래를 반복해 주세요
    num = int(input())             # 입력받은 글자를 정수로 바꿔 num 에 담아 주세요
    if num > max:                  # 만약 num 이 max 보다 크면
        print(num, 'is too big!')  # num 과 'is too big!' 를 화면에 보여 주세요
        break                      # 반복을 멈추고 빠져나와 주세요

if / elif / else 순서

>>> c = 15 * 5                        # c 에 15 * 5 를 담아 주세요
>>> d = 15 + 15 + 15 + 15 + 15        # d 에 15 + 15 + 15 + 15 + 15 를 담아 주세요
>>> if c > d:                         # 만약 c 가 d 보다 크면
...     print('c is greater than d')  # 'c is greater than d' 를 화면에 보여 주세요
... elif c == d:                      # 아니고, 만약 c 와 d 가 같으면
...     print('c is equal to d')      # 'c is equal to d' 를 화면에 보여 주세요
... else:                             # 그 밖의 경우에는
...     print('I don\'t know')        # 'I don\'t know' 를 화면에 보여 주세요
...

중첩 if

year = int(input())               # 입력받은 글자를 정수로 바꿔 year 에 담아 주세요

if year % 4 == 0:                 # 만약 year 를 4 로 나눈 나머지가 0 이면
    if year % 100 == 0:           # 만약 year 를 100 으로 나눈 나머지가 0 이면
        if year % 400 == 0:       # 만약 year 를 400 으로 나눈 나머지가 0 이면
            is_leap_year = True   # is_leap_year 에 True 를 담아 주세요
        else:                     # 그 밖의 경우에는
            is_leap_year = False  # is_leap_year 에 False 를 담아 주세요
    else:                         # 그 밖의 경우에는
        is_leap_year = True       # is_leap_year 에 True 를 담아 주세요
else:                             # 그 밖의 경우에는
    is_leap_year = False          # is_leap_year 에 False 를 담아 주세요
