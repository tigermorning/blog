무한 루프와 break

# filename: ten.py
max = 10                           # 기준값 10 을 max 에 담아요

while True:                        # 조건이 항상 참이라 계속 반복해요(무한 루프)
    num = int(input())             # 입력받은 글자를 정수로 바꿔 num 에 담아요
    if num > max:               # num 이 max 보다 크면 실행해요
        print(num, 'is too big!')  # num 과 너무 크다는 메시지를 출력해요
        break                      # 반복을 즉시 끝내고 빠져나가요

if / elif / else 순서

>>> c = 15 * 5                  # 15 곱하기 5 인 75 를 c 에 담아요
>>> d = 15 + 15 + 15 + 15 + 15  # 15 를 다섯 번 더한 75 를 d 에 담아요
>>> if c > d:                # 위에서부터 검사해 처음 참인 것 하나만 실행해요
...     print('c is greater than d')     # c 가 더 클 때 출력해요
... elif c == d:                         # 앞이 거짓이고 c 와 d 가 같으면 실행해요

... else:                                # 위가 모두 거짓이면 실행해요

중첩 if

year = int(input())               # 입력한 연도를 정수로 받아 year 에 담아요

if year % 4 == 0:                 # 4 로 나누어떨어지면 실행해요
    if year % 100 == 0:           # 100 으로도 나누어떨어지면 실행해요
        if year % 400 == 0:       # 400 으로도 나누어떨어지면 실행해요
            is_leap_year = True   # 윤년이라서 True 를 담아요
        else:                     # 400 으로는 안 나누어떨어지면 실행해요
            is_leap_year = False  # 윤년이 아니라서 False 를 담아요
    else:                         # 100 으로는 안 나누어떨어지면 실행해요
        is_leap_year = True       # 윤년이라서 True 를 담아요
else:                             # 4 로 나누어떨어지지 않으면 실행해요
    is_leap_year = False          # 윤년이 아니라서 False 를 담아요
