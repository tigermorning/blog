while

>>> num = 1
>>> while num <= 3:    # 참인 동안 반복
...     print(num)
...     num = num + 1    # 조건의 값을 바꾸는 줄이 있어야 끝난다

무한 루프와 break

# filename: ten.py
max = 10

while True:              # 무한 루프
    num = int(input())
    if num > max:
        print(num, 'is too big!')
        break            # 즉시 탈출

if / elif / else 순서

>>> c = 15 * 5
>>> d = 15 + 15 + 15 + 15 + 15
>>> if c > d:                  # 위에서부터 검사, 처음 참인 것 하나만 실행
...     print('c is greater than d')
... elif c == d:
...     print('c is equal to d')
... else:
...     print('I don\'t know')
...
c is equal to d

핵심 한 줄: <=는 작은 경계부터, >=는 큰 경계부터.

중첩 if

year = int(input())

if year % 4 == 0:
    if year % 100 == 0:
        if year % 400 == 0:
            is_leap_year = True
        else:
            is_leap_year = False
    else:
        is_leap_year = True
else:
    is_leap_year = False

2000 -> 2000 is a leap year
1900 -> 1900 is not a leap year
2024 -> 2024 is a leap year
2023 -> 2023 is not a leap year
